### أ. هيكل البيانات في قاعدة البيانات (`Backend Schema in apps.catalog`):
```python
# نموذج توسعة بيانات العطور في apps/catalog/models.py
class PerfumeAttribute(models.Model):
    product = models.OneToOneField(Product, on_delete=models.CASCADE, related_name="perfume_details")
    fragrance_family = models.CharField(max_length=100) # شرقي، خشبي، أروماتيك، حمضي، زهري
    gender = models.CharField(max_length=50, choices=[("MEN", "رجالي"), ("WOMEN", "نسائي"), ("UNISEX", "للجنسين")])
    
    # الهرم العطري (JSON Fields للمكونات وأيقوناتها)
    top_notes = models.JSONField(default=list)    # [{"name": "برغموت", "icon": "citrus", "desc": "افتتاحية منعشة"}]
    heart_notes = models.JSONField(default=list)  # [{"name": "عود كمبودي", "icon": "oud", "desc": "قلب دافئ وفاخر"}]
    base_notes = models.JSONField(default=list)   # [{"name": "مسك أبيض", "icon": "musk", "desc": "ثبات طويل وأنيق"}]
    
    # مؤشرات الأداء (قيم رقمية من 1 إلى 5)
    longevity_score = models.PositiveSmallIntegerField(default=4) # درجة الثبات
    longevity_hours = models.CharField(max_length=50, default="12-16 ساعة")
    sillage_score = models.PositiveSmallIntegerField(default=4)   # درجة الفوحان (قريب / معتدل / قوي / هائل)
    
    # التصنيف والفصول
    seasons = models.JSONField(default=list)     # ["winter", "autumn"]
    occasions = models.JSONField(default=list)   # ["formal", "evening", "special_dates"]
    concentration = models.CharField(max_length=50, default="Eau de Parfum") # Parfum, Extrait, EDP, EDT
    origin_country = models.CharField(max_length=100, default="فرنسا")
```
